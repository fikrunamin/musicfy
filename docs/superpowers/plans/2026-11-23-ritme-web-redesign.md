# Ritme Web Redesign (#2c) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Restyle and restructure the Ritme web companion (`ritme-web/`) to the designer's handoff (`docs/design/handoff/ritme-web-handoff.html`, flows A–P), without changing logic or behaviour.

**Architecture:** Only the UI layer changes. The single `src/styles.css` becomes a set of token-driven stylesheets under `src/styles/` (the old rules live on in `legacy.css` until each screen replaces them). Shared Preact components (`components.tsx`, `media.tsx`, `sheet.tsx`, `icon.tsx`) are built first, and the screens are then rebuilt in flow order on top of them. The YouTube player keeps one DOM container that occupies the artwork slot (the "one stage" of brief §11), so no layout change ever re-creates the iframe.

**Tech Stack:** Preact 10.29 + Vite 7 + TypeScript 5.9, Vitest 4 (jsdom) + @testing-library/preact, Playwright 1.65 (e2e and screenshots against the real Ritme Ktor server). Inter is self-hosted (woff2, SIL OFL), and icons are inline SVG paths from Material Symbols Rounded (Apache-2.0). No new runtime dependency.

**Phases:** Phase 1 (Tasks 1–6, full detail below) covers the foundations, the shared components and screens A, B, C, D and E. Phase 2 (F, G, H, the C+E desktop room, motion) and Phase 3 (video and direct-link screens I–P, final polish) are outlined at the end and will be detailed later.

 * No real network beyond the local server: YouTube is faked, search and video info are canned,
 * the rest blocked. `songs` is what every search answers.
 */
export async function isolate(context: BrowserContext, songs: CannedSong[] = [SONG]) {
  // Routes registered later win, so the catch-all goes first.
  await context.route(/^https?:\/\/(?!localhost[:/])/, (route) => route.abort());
  await context.route('https://www.youtube.com/iframe_api', (route) =>
    route.fulfill({ contentType: 'text/javascript', body: FAKE_IFRAME_API }),
  );
  await context.route('**/api/search?**', (route) => route.fulfill({ json: songs }));
  await context.route('**/api/video?**', (route) => route.fulfill({ json: VIDEO }));
}

/** Closes the open sheet, if any, through its close button. */
export async function closeSheet(page: Page) {
  const dialog = page.getByRole('dialog');
  if ((await dialog.count()) === 0) return;
  await dialog.getByRole('button', { name: 'Tutup' }).click();
  await expect(dialog).toHaveCount(0);
}

/** Brings up "Tambah ke antrean": the inline panel, or the sheet behind "Tambah lagu" on a phone. */
export async function showAdd(page: Page) {
  if ((await page.getByLabel('Cari lagu').count()) === 0) await page.getByRole('button', { name: 'Tambah lagu' }).first().click();
  await expect(page.getByLabel('Cari lagu')).toBeVisible();
}

/** The members list: the inline panel, or the sheet behind the avatar stack on a phone. */
export async function showMembers(page: Page): Promise<Locator> {
  const opener = page.getByRole('button', { name: /^\d+ anggota$/ });
  if ((await opener.count()) > 0) {
    await opener.click();
    const sheet = page.getByRole('dialog', { name: 'Anggota' });
    await expect(sheet).toBeVisible();
    return sheet;
  }
  return page.locator('section[aria-labelledby="members-title"]');
}

/** Waits until `page` lists `name` as a room member. */
export async function expectMember(page: Page, name: string) {
  const members = await showMembers(page);
  await expect(members.locator('.members')).toContainText(name);
  await closeSheet(page);
}

/** Searches and adds one song to the end of the queue. */
export async function addSong(page: Page, query: string, title: string) {
  await showAdd(page);
  await page.getByLabel('Cari lagu').fill(query);
  await page.getByRole('button', { name: 'Cari', exact: true }).click();
  await page.getByRole('button', { name: `Tambah: ${title}` }).click();
  await closeSheet(page);
}

/** Tab A starts a room as Budi; tab B joins it through /r/{code} as Sari. */
export async function openRoom(context: BrowserContext): Promise<{ a: Page; b: Page; code: string }> {
  const a = await context.newPage();
  await a.goto('/');
  await a.getByLabel('Nama tampilan').fill('Budi');
  await a.getByRole('button', { name: 'Mulai Ritme' }).click();
  const heading = a.getByRole('heading', { level: 1 });
  await expect(heading).toHaveText(/^Ritme · [2-9A-HJ-NP-Z]{6}$/);
  const code = (await heading.textContent())!.slice(-6);
  await expect(a).toHaveURL(new RegExp(`/r/${code}$`));

  const b = await context.newPage();
  await b.goto(`/r/${code}`);
  await expect(b.getByRole('link', { name: 'Buka di aplikasi Musicfy' })).toHaveAttribute('href', `musicfy://ritme/${code}`);
  await b.getByLabel('Nama tampilan').fill('Sari');
  await b.getByRole('button', { name: 'Gabung' }).click();
  await expect(b.getByRole('heading', { level: 1 })).toHaveText(`Ritme · ${code}`);
  await expectMember(a, 'Sari');
  return { a, b, code };
}

import type { ComponentChildren } from 'preact';
import { httpsThumbnail } from '../net/search';
import { DurationChip, MetaClip, cx } from './components';
import { Icon } from './icon';

export type ThumbKind = 'song' | 'video' | 'direct';

interface ThumbProps {
  url: string | null;
  /** 78 × 44 (16:9) instead of 44 × 44: video rows, and every row of a queue that holds videos. */
  wide?: boolean;
  /** Which placeholder to show without an image: a note, a play mark, or the film of a direct link. */
  kind?: ThumbKind;
  /** Shown as a chip on wide thumbnails (videos). */
  duration?: string | null;
}

/**
 * A thumbnail (handoff "Baris Lagu" / "Baris Video"): 44 × 44 or 78 × 44, r8, or a placeholder
 * unless the URL is https (entries come from any client).
 */
export function Thumb({ url, wide = false, kind = 'song', duration = null }: ThumbProps) {
  const src = httpsThumbnail(url);
  const cls = cx('thumb', wide && 'thumb-wide');
  const image = src ? (
    <img class={cls} src={src} alt="" loading="lazy" referrerpolicy="no-referrer" width={wide ? 78 : 44} height={44} />
  ) : (
    <div class={cx(cls, 'thumb-empty', kind === 'direct' && 'thumb-direct')} aria-hidden="true">
      <Icon name={kind === 'song' ? 'music_note' : kind === 'direct' ? 'movie' : 'play_arrow'} size={20} />
    </div>
  );
  if (!wide || !duration) return image;
  return (
    <span class="thumb-frame">
      {image}
      <DurationChip>{duration}</DurationChip>
    </span>
  );
}

/**
 * The dark 16:9 frame for a direct link without a thumbnail (handoff "Placeholder Video"): film
 * icon and domain, with an optional badge (top left) and duration (bottom right).
 */
export function VideoPlaceholder({ domain, badge, duration }: { domain: string; badge?: ComponentChildren; duration?: string | null }) {
  return (
    <div class="video-placeholder">
      {badge && <span class="vp-badge">{badge}</span>}
      <Icon name="movie" size={32} />
      <span class="vp-domain">{domain}</span>
      {duration && (
        <span class="vp-duration">
          <DurationChip>{duration}</DurationChip>
        </span>
      )}
    </div>
  );
}

  thumb: ComponentChildren;
  title: string;
  /** The one-line meta; replaced by `warning` (danger colour) when the entry cannot play here. */
  meta: ComponentChildren;

  trailing?: ComponentChildren;
  /** Content under the row (the inline actions of an opened row). */
  below?: ComponentChildren;
  current?: boolean;
}

/** A 60 px song or video row: thumbnail, title, one-line meta, trailing control. */
export function MediaRow({ thumb, title, meta, warning, trailing, below, current = false }: RowProps) {
  return (

  );
}

import type { ComponentChildren, RefObject } from 'preact';
import { useEffect, useId, useRef } from 'preact/hooks';
import { Button } from './components';
import { t } from './strings';

const FOCUSABLE = 'a[href], button:not([disabled]), input:not([disabled]), select, textarea, [tabindex]:not([tabindex="-1"])';

function focusables(root: HTMLElement): HTMLElement[] {
  return [...root.querySelectorAll<HTMLElement>(FOCUSABLE)].filter((el) => !el.closest('[hidden], [inert]'));
}

/**
 * Modal focus handling for sheets and dialogs: focus moves in (to `initial`, else the first
 * control), Tab wraps inside, Escape closes, the page behind does not scroll, and focus goes back
 * to whatever opened it.
 */
export function useModal(panel: RefObject<HTMLElement>, onClose: () => void, initial?: RefObject<HTMLElement>) {
  const close = useRef(onClose);
  close.current = onClose;
  useEffect(() => {
    const opener = document.activeElement as HTMLElement | null;
    const root = panel.current!;
    (initial?.current ?? focusables(root)[0] ?? root).focus();
    document.body.classList.add('modal-open');
    // A dialog opened inside a sheet handles its own keys first and stops them there.
    const onKey = (e: KeyboardEvent) => {
      if (e.key !== 'Escape' && e.key !== 'Tab') return;
      e.stopPropagation();
      if (e.key === 'Escape') {
        close.current();
        return;
      }
      const items = focusables(root);
      if (items.length === 0) return;
      const first = items[0];
      const last = items[items.length - 1];
      if (e.shiftKey && document.activeElement === first) {
        e.preventDefault();
        last.focus();
      } else if (!e.shiftKey && document.activeElement === last) {
        e.preventDefault();
        first.focus();
      }
    };
    root.addEventListener('keydown', onKey);
    return () => {
      root.removeEventListener('keydown', onKey);
      document.body.classList.remove('modal-open');
      if (opener?.isConnected) opener.focus();
    };
  }, []);
}

import { useState } from 'preact/hooks';
import type { KeyValueStore } from '../net/storage';
import { NAME_KEY } from '../net/storage';
import type { SessionState } from '../session/reducer';
import type { SessionApi } from '../session/session';

interface Props {
  session: SessionApi;
  state: SessionState;
  /** From /r/{code}: the invite page (flow B), joining that room. */
  routeCode: string | null;
  prefs: KeyValueStore;
}

type FieldErrorState = { field: 'name' | 'code'; message: string } | null;

/** The vinyl and sleeve of flows A/B, drawn in CSS (monochrome: no artwork before a room). */

  );
}

export function HomeScreen({ session, state, routeCode, prefs }: Props) {
  const [name, setName] = useState(prefs.get(NAME_KEY) ?? '');
  const [code, setCode] = useState(routeCode ?? '');
  const [error, setError] = useState<FieldErrorState>(null);
  const busy = state.phase === 'connecting';
  const ended = state.phase === 'ended' && state.ended ? endText(state.ended) : null;
  const invite = routeCode !== null;

  const validName = (): string | null => {
    const n = normalizeName(name);
    if (!n) setError({ field: 'name', message: t.nameInvalid });
    else prefs.set(NAME_KEY, n);
    return n;
  };

  const onCreate = () => {
    setError(null);
    const n = validName();
    if (!n) return;
    session.dismiss();
    session.create(n);
  };

  const onJoin = (e: Event) => {
    e.preventDefault();
    setError(null);
    const n = validName();
    if (!n) return;
    const c = invite ? routeCode : normalizeCode(code);
    if (!c) {
      setError({ field: 'code', message: t.codeInvalid });
      return;
    }
    session.dismiss();
    session.join(c, n);
  };

  const nameError = error?.field === 'name' ? 'name-error' : null;
  const codeError = error?.field === 'code' ? 'code-error' : null;

  const nameField = (
    <>
      <TextField
        id="name-input"
        name="name"
        label={t.nameLabel}
        value={name}
        maxLength={32}
        autoComplete="nickname"
        placeholder={t.namePlaceholder}
        errorId={nameError}
        onInput={(e) => setName(e.currentTarget.value)}
      />
      {nameError && <FieldError id={nameError}>{error!.message}</FieldError>}
    </>
  );

  return (
    <main class={cx('home', invite && 'home-invite')}>
      <HeroArt />
      <div class="home-panel">
        {invite ? (
          <p class="room-chip">
            <Icon name="group" size={20} />
            {t.roomChip(routeCode)}
          </p>
        ) : (
          <p class="home-brand">
            <Icon name="radio_button_checked" size={22} />
            <span class="home-brand-mobile">{t.brand}</span>
            <span class="home-brand-desktop">{t.brandRitme}</span>
          </p>
        )}
        <header class="home-header">
          <h1>{invite ? t.invited : t.appName}</h1>
          <p class="home-tagline">{t.tagline}</p>

  /** What the slot shows while there is no player to show: the cover, or the empty placeholder. */
  cover: ComponentChildren;
  /** Shown in the slot when the current entry cannot play in this browser. */
  unplayableSlot: ComponentChildren;
  /** The controller asks for (or no longer needs) a tap on the player itself. */
  onNeedsTap: (needsTap: boolean) => void;
}

/**
 * The visible player (never hidden behind anything, per YouTube's terms); exists only after the
 * tap gate. Its container stays mounted while the stage changes shape, so the iframe is never
 * re-created by a layout change.
 */
function PlayerHost({ session, room, unplayable, pageIsHttps, cover, unplayableSlot, onNeedsTap }: StageProps) {
  const container = useRef<HTMLDivElement>(null);
  const controller = useRef<PlaybackController | null>(null);

  useEffect(() => {
    const el = container.current!;
    const c = new PlaybackController({
      createPlayer: (kind, onEvent) => (kind === 'youtube' ? createYouTubePlayer(el, onEvent) : createVideoPlayer(el, onEvent)),
      serverNow: () => session.serverNow(),
      pageIsHttps,
      onUnplayable: (entryId) => session.markUnplayable(entryId),
      onNeedsTap,
    });
    controller.current = c;
    c.update(room, unplayable);
    // The session decides when: on becoming visible, or once the clock is fresh after a sleep or reconnect.
    const offResync = session.onResync(() => c.resync());
    const release = keepScreenOn();
    return () => {
      offResync();
      release();
      c.destroy();
      controller.current = null;
      onNeedsTap(false);
    };
  }, [session, pageIsHttps]);

  useEffect(() => {
    controller.current?.update(room, unplayable);
  }, [room, unplayable]);

  const entry = currentEntry(room);
  const verdict = entry ? playability(entry, pageIsHttps) : null;
  const hidden = !verdict?.playable || (entry !== null && unplayable[entry.entryId] === true);

  return (
    <>
      <div ref={container} class="player" hidden={hidden} />
      {hidden && (entry ? unplayableSlot : cover)}
    </>
  );
}

/**
 * The listen-mode content of the stage (brief §11): before the first tap the cover with
 * "Ketuk untuk ikut dengar" (browsers need one gesture before sound); after it, the player in the
 * same slot with nothing on top. Leaving listen mode unmounts this, so coming back asks again.
 */
export function ListenStage(props: StageProps) {
  const [gateOpen, setGateOpen] = useState(false);

  // Fetch the IFrame API early so the player appears quickly after the tap.
  useEffect(() => {
    loadYouTubeApi().catch(() => {});
  }, []);

  if (gateOpen) return <PlayerHost {...props} />;
  return (
    <>
      {props.cover}
      <div class="stage-gate">
        <button type="button" class="btn btn-primary gate-btn" onClick={() => setGateOpen(true)}>
          <Icon name="headphones" size={22} />
          <span class="btn-label">{t.tapToListen}</span>
        </button>
        <p class="gate-reason">{t.tapReason}</p>
      </div>
    </>
  );
}

import { useState } from 'preact/hooks';

interface Props {

  room: RoomState;
  memberId: string | null;
  unplayable: Readonly<Record<string, true>>;
  online: boolean;
  pageIsHttps: boolean;
  /** upcoming: "Berikutnya" and "Sudah diputar" (mobile, under the player); full: also "Sedang diputar" (desktop column). */
  variant?: 'upcoming' | 'full';
  /** Opens the add sheet (or focuses the add panel); omitted while the member may not add. */
  onAdd?: () => void;
}

interface RowProps extends Props {
  entry: QueueEntry;
  upcoming: boolean;
  current: boolean;
  /** The queue holds videos: every thumbnail is 78 × 44 (DECISIONS 4). */
  mixed: boolean;
  open: boolean;
  onToggle: () => void;
}

function Row({ entry, session, room, memberId, unplayable, online, pageIsHttps, upcoming, current, mixed, open, onToggle }: RowProps) {
  const control = online && canControl(room, memberId);
  const removable = online && canRemove(room, memberId, entry);
  const up = upcoming && control ? moveUpTarget(room, entry.entryId) : null;
  const down = upcoming && control ? moveDownTarget(room, entry.entryId) : null;
  const label = unplayableLabel(entry, unplayable, pageIsHttps);
  const video = isVideoEntry(entry);
  const live = isLive(entry);
  const by = t.addedBy(memberName(room.members, entry.addedBy, memberId));
  const hasActions = !current && (control || removable);
  const actionsId = `row-actions-${entry.entryId}`;
  // A video's badge says Video or LIVE and its length is on the thumbnail; a song's meta says LIVE.
  const meta = video ? (
    <>
      <Badge kind={live ? 'live' : 'video'} />
      <span>{[entrySource(entry), by].join(' · ')}</span>
    </>
  ) : (
    <span>{[entrySource(entry), live ? t.live : null, by].filter(Boolean).join(' · ')}</span>
  );

  return (
    <MediaRow
      current={current}
      thumb={
        <Thumb
          url={entry.thumbnailUrl}
          wide={mixed}
          kind={entry.kind === 'URL' ? 'direct' : video ? 'video' : 'song'}
          duration={video && !live ? formatMs(entry.durationMs) : null}
        />
      }
      title={entry.title}
      meta={meta}
      warning={label}
      trailing={
        current ? (
          <EqBars playing={room.playback.isPlaying} />
        ) : (
          hasActions && (
            <IconButton icon="more_vert" label={t.rowActions(entry.title)} aria-expanded={open} aria-controls={actionsId} onClick={onToggle} />
          )
        )
      }
      below={
        open &&
        hasActions && (
          <div id={actionsId} class="row-actions">
            {control && (
              <Button icon="play_arrow" class="row-play" aria-label={`${t.playNow}: ${entry.title}`} onClick={() => session.playEntry(entry.entryId)}>
                {t.playNow}
              </Button>
            )}
            {up && (
              <IconButton icon="arrow_upward" variant="filled" label={`${t.moveUp}: ${entry.title}`} onClick={() => session.move(entry.entryId, up.afterEntryId)} />
            )}
            {down && (
              <IconButton
                icon="arrow_downward"
                variant="filled"
                label={`${t.moveDown}: ${entry.title}`}
                onClick={() => session.move(entry.entryId, down.afterEntryId)}
              />
            )}
            {removable && (
              <IconButton icon="delete" variant="danger" label={`${t.remove}: ${entry.title}`} onClick={() => session.remove(entry.entryId)} />
            )}
          </div>
        )
      }
    />
  );
}

/** "Antrean kosong." with the vinyl mark and, for members who may add, the way to start. */
function EmptyQueue({ onAdd }: { onAdd?: () => void }) {
  return (
    <div class="empty-queue">
      <span class="empty-mark">
        <Icon name="album" size={40} />
      </span>
      <p class="empty-title">{t.queueEmpty}</p>

        <Button variant="primary" icon="add" onClick={onAdd}>
          {t.addSong}
        </Button>
      )}
    </div>
  );
}

export function QueueList(props: Props) {
  const { room, variant = 'upcoming', onAdd } = props;
  const { played, current, upcoming } = splitQueue(room);
  const [openId, setOpenId] = useState<string | null>(null);

  const row = (entry: QueueEntry, kind: 'current' | 'upcoming' | 'played') => (

      key={entry.entryId}
      {...props}
      entry={entry}
      current={kind === 'current'}
      upcoming={kind === 'upcoming'}
      mixed={mixed}

      onToggle={() => setOpenId(openId === entry.entryId ? null : entry.entryId)}
    />
  );

      <section class={cx('queue-panel', `queue-${variant}`)} aria-labelledby="queue-title">
        <h2 id="queue-title" class="visually-hidden">
          {full ? t.queue : t.upNext}
        </h2>
        <EmptyQueue onAdd={onAdd} />
      </section>
    );

    <section class={cx('queue-panel', `queue-${variant}`)} aria-labelledby="queue-title">

          <ul class="rows">{row(current, 'current')}</ul>
        </>

      {full && <h3 class="queue-section">{t.upNext}</h3>}
      {upcoming.length === 0 ? <p class="queue-none">{t.queueEmpty}</p> : <ul class="rows">{upcoming.map((e) => row(e, 'upcoming'))}</ul>}
      {played.length > 0 && (
        <div class="played">
          <button type="button" class="played-toggle" aria-expanded={showPlayed} aria-controls="played-list" onClick={() => setShowPlayed(!showPlayed)}>

            <ul id="played-list" class="rows">
              {played.map((e) => row(e, 'played'))}
            </ul>
          )}
        </div>
      )}
    </section>
  );
}
